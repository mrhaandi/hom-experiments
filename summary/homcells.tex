\section{Overview}

The aim is an alternative proof of decidability of higher-order matching,
in the spirit of Padovani's cells.  A matching problem asks for a closed
term $t$ with $t\,\vec s_q=_\beta b_q$ for finitely many given closed
argument tuples $\vec s_q$ and observable values $b_q$.  The notes are
organised along the following chain of reductions.

\begin{enumerate}
\item \textbf{Observations.}  Contextual equivalence is parametrized by a
  set of observations $\mathcal M$; a matching problem is a finite
  applicative $\mathcal M$ (Section~\ref{sec:obs}).
\item \textbf{A finite model.}  Observable behaviour is exactly the value
  in a finite \emph{truncation model}; for finite applicative $\mathcal M$,
  $\ObsEq_{\mathcal M}$ is equality of $\sem t$ at finitely many points
  (Theorem~\ref{thm:semchar}).
\item \textbf{Stretching.}  Repeated layers of a term can be removed or
  inserted without changing observations (Section~\ref{sec:stretch}).
\item \textbf{Cells.}  Terms with a bounded number of \emph{slots} for
  variables are trees over a finite alphabet, and their semantics is a
  finite tree automaton.  For fixed slot numbers, existence of a cellular
  solution is decidable (Corollary~\ref{cor:cellular-decidable}).
\item \textbf{Semi-cellular terms.}  A semi-cellular term is
  $\ObsEq_{\mathcal M}$ to a cellular one (Lemma~\ref{lem:semicell}).
\item \textbf{Main conjecture.}  Every term is semi-cellular for slot
  numbers computable from the problem (Conjecture~\ref{conj:main}).
  Steps 1--5 are proved; the conjecture is the open core.
  Section~\ref{sec:bounding} develops tools for it: merging variables
  using representatives, and contexts as transducers of intersection-type
  requirements.
\end{enumerate}

\section{Setting and the truncation model}\label{sec:setting}

We work in the simply typed $\lambda$-calculus with one ground type
$\Ground$ over a finite signature of ground constants $a,b,\ldots$ and
first-order symbols $f$ of arity $\arity(f)$.  We only \emph{observe}
ground normal forms of depth at most $1$, i.e.\ first-order terms
$f(a_1,\ldots,a_r)$ with $r=\arity(f)$ and ground constants $a_j$
(constants being the case $r=0$).  The set of observable values of depth
$0$ (constants) is written $\Val^0$, and of depth at most $1$ is written
$\Val^1$.  Deeper ground normal forms, such as $f(f(a))$, exist but are
not observable.

All terms are in $\eta$-long form; $\lambda x_1\ldots x_n.u$ abbreviates
$\lambda x_1.(\cdots(\lambda x_n.u)\cdots)$ with $u$ of ground type.
$\NF(u)$ denotes the $\beta$-normal form of $u$.

\begin{definition}[Truncation model]
Let $D_\Ground=\Val^1\cup\{\bot\}$ and $D_{A\to B}=D_B^{\,D_A}$ (all
functions).  Interpret a constant $a$ as $a$, and a first-order symbol $f$
of arity $r$ as
\[
  \hat f(d_1,\ldots,d_r)=
  \begin{cases}
    f(d_1,\ldots,d_r) & \text{if } d_1,\ldots,d_r\in\Val^0,\\
    \bot & \text{otherwise.}
  \end{cases}
\]
For a term $s$ and an environment $\eta$ (mapping the free variables of
$s$ of type $B$ to $D_B$), $\sem{s}_\eta$ is defined as usual; for closed
$s$ we write $\sem{s}$.  Each $D_A$ is finite.
\end{definition}

This is a model of the simply typed $\lambda$-calculus with constants, so
$s=_\beta s'$ implies $\sem{s}_\eta=\sem{s'}_\eta$, and $\sem{\cdot}$ is
compositional: $\sem{C[s]}$ depends on a closed $s$ only through $\sem{s}$.

\begin{lemma}[Adequacy]\label{lem:adequacy}
For every closed $N:\Ground$, $\sem N=\NF(N)$ if $\NF(N)\in\Val^1$, and
$\sem N=\bot$ otherwise.
\end{lemma}
\begin{proof}
By $\beta$-soundness we may assume $N$ normal.  A closed normal term of
type $\Ground$ has a constant head, so it is a first-order ground term.
Induction on $N$.  If $N=a$, then $\sem{N}=a\in\Val^0$.  If
$N=f(N_1,\ldots,N_r)$, then by the induction hypothesis
$\sem{N_j}\in\Val^0$ iff $N_j$ is a constant.  If all $N_j$ are
constants, $\sem{N}=N\in\Val^1$.  Otherwise some $\sem{N_j}\notin\Val^0$,
so $\sem{N}=\bot$, and $N$ has depth at least $2$, so $N\notin\Val^1$.
\end{proof}

\section{Contextual equivalence under observations}\label{sec:obs}

Throughout, $i\in\{0,1\}$ is fixed and $A$ is a type.

\begin{definition}[Observations]
A \emph{set of observations at type $A$} is a set $\mathcal M$ of
contexts $C[-]$ such that, for every closed $t:A$, the term $C[t]$ is
closed and of type $\Ground$.  We write $\mathcal M^{\mathrm{all}}_A$ for
the set of all such contexts.  An observation is \emph{applicative} if it
has the form $C[-]=[-]\,s_1\ldots s_n$ with $s_1,\ldots,s_n$ closed; we
then identify $C$ with the tuple $\vec s$.
\end{definition}

\begin{definition}[Contextual preorder and equivalence under $\mathcal M$]
For closed $t,t':A$ we write $t\ObsLeq^i_{\mathcal M}t'$ if for every
$C\in\mathcal M$
\[
        \NF(C[t])\in\Val^i
        \quad\Longrightarrow\quad
        \NF(C[t])=\NF(C[t']),
\]
and $t\ObsEq^i_{\mathcal M}t'$ when $t\ObsLeq^i_{\mathcal M}t'$ and
$t'\ObsLeq^i_{\mathcal M}t$.  For $\mathcal M=\mathcal M^{\mathrm{all}}_A$
we write simply $\ObsLeq^i$ and $\ObsEq^i$.
\end{definition}

Thus $t\ObsEq^i_{\mathcal M}t'$ says: on every $C\in\mathcal M$ either
both $C[t]$, $C[t']$ are unobservable, or both yield the same value of
$\Val^i$.  A matching problem $\{t\,\vec s_q=_\beta b_q\}_q$ is the
condition $t\ObsEq^1_{\mathcal M}t_0$ for the finite applicative
$\mathcal M=\{\vec s_q\}_q$ and any $t_0$ with these values.

\begin{lemma}[Basic properties]\label{lem:basic}
\begin{enumerate}[label=(\alph*)]
\item $\ObsLeq^i_{\mathcal M}$ is a preorder and $\ObsEq^i_{\mathcal M}$ is
  an equivalence relation.
\item If $\mathcal M_1\subseteq\mathcal M_2$, then
  $t\ObsLeq^i_{\mathcal M_2}t'$ implies $t\ObsLeq^i_{\mathcal M_1}t'$.
\item ${\ObsLeq^i_{\bigcup_j\mathcal M_j}}=\bigcap_j{\ObsLeq^i_{\mathcal M_j}}$,
  and likewise for $\ObsEq^i$.
\item If $t=_\beta t'$, then $t\ObsEq^i_{\mathcal M}t'$.
\item $\ObsLeq^1_{\mathcal M}\subseteq\ObsLeq^0_{\mathcal M}$.
\end{enumerate}
\end{lemma}
\begin{proof}
(a) For transitivity let $t\ObsLeq^i_{\mathcal M}t'\ObsLeq^i_{\mathcal M}t''$
and $C\in\mathcal M$ with $\NF(C[t])\in\Val^i$.  Then
$\NF(C[t'])=\NF(C[t])\in\Val^i$, hence $\NF(C[t''])=\NF(C[t])$.  (b),
(c) hold since the definition is a conjunction over $C\in\mathcal M$.
(d) Normal forms are unique.  (e) $\Val^0\subseteq\Val^1$.
\end{proof}

\begin{lemma}[Congruence]\label{lem:congruence}
Let $D[-]$ be a context such that $D[t]$ is closed of type $B$ for closed
$t:A$, let $\mathcal M$ be a set of observations at type $B$, and put
$\mathcal M\circ D=\{\,C[D[-]]\mid C\in\mathcal M\,\}$.  If
$t\ObsLeq^i_{\mathcal M\circ D}t'$, then $D[t]\ObsLeq^i_{\mathcal M}D[t']$;
likewise for $\ObsEq^i$.  In particular $\ObsEq^i$ is a congruence.
\end{lemma}
\begin{proof}
$C[D[t]]=(C\circ D)[t]$; for the second part note
$\mathcal M^{\mathrm{all}}_B\circ D\subseteq\mathcal M^{\mathrm{all}}_A$.
\end{proof}

For an observation $C$ at type $A$ let $\sem{C}=\sem{\lambda g.C[g]}\in
D_{A\to\Ground}$.  Since $t$ is closed, $C[t]=_\beta(\lambda g.C[g])\,t$,
so $\sem{C[t]}=\sem{C}(\sem{t})$.  Let $\pi:D_\Ground\to D_\Ground$ send
$\Val^1\setminus\Val^0$ to $\bot$ and fix the rest, and write
$d\sqsubseteq d'$ for the flat order ($d=\bot$ or $d=d'$).

\begin{theorem}[Semantic characterization]\label{thm:semchar}
For closed $t,t':A$ and a set of observations $\mathcal M$ at type $A$:
\begin{enumerate}[label=(\alph*)]
\item $t\ObsLeq^1_{\mathcal M}t'$ iff
  $\sem{C}(\sem t)\sqsubseteq\sem{C}(\sem{t'})$ for every $C\in\mathcal M$;
\item $t\ObsEq^1_{\mathcal M}t'$ iff
  $\sem{C}(\sem t)=\sem{C}(\sem{t'})$ for every $C\in\mathcal M$;
\item $t\ObsEq^0_{\mathcal M}t'$ iff
  $\pi(\sem{C}(\sem t))=\pi(\sem{C}(\sem{t'}))$ for every $C\in\mathcal M$.
\end{enumerate}
In particular $\sem t=\sem{t'}$ implies $t\ObsEq^i_{\mathcal M}t'$ for
every $\mathcal M$.  For applicative $\mathcal M=\{\vec s_1,\ldots,\vec s_p\}$,
$t\ObsEq^1_{\mathcal M}t'$ iff $\sem t(\sem{\vec s_q})=\sem{t'}(\sem{\vec s_q})$
for all $q$; so a matching problem is the condition
$\sem t(\sem{\vec s_q})=b_q$, with computable points $\sem{\vec s_q}$.
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:adequacy}, $\NF(C[t])\in\Val^1$ iff
$\sem{C[t]}\neq\bot$, and then $\NF(C[t])=\sem{C[t]}$.  Unfolding the
definitions gives (a) and (b) for a single $C$; for $\mathcal M$ use
Lemma~\ref{lem:basic}(c).  For (c), $\NF(C[t])\in\Val^0$ iff
$\pi(\sem{C[t]})\neq\bot$.
\end{proof}

Equality in the model is only a \emph{sufficient} condition for
$\ObsEq^1$ over all contexts, because $D$ contains non-definable elements.

\begin{lemma}[Context lemma]\label{lem:context}
For closed $t,t':A_1\to\cdots\to A_n\to\Ground$,
$t\ObsLeq^i t'$ iff $t\ObsLeq^i_{\mathcal M^{\mathrm{app}}_A}t'$, where
$\mathcal M^{\mathrm{app}}_A$ is the set of all applicative observations.
\end{lemma}
\begin{proof}[Proof idea]
($\Rightarrow$) by Lemma~\ref{lem:basic}(b).  ($\Leftarrow$) is Milner's
context lemma, by induction on the length of the head reduction of
$C[t]$.  \textbf{TODO:} check the preorder form with values restricted to
$\Val^i$.
\end{proof}

\section{Stretching}\label{sec:stretch}

\subsection{The concrete stretching lemma}

\begin{proposition}[Stretching lemma]\label{prop:stretch}
Assume the signature has exactly one first-order symbol $f$, of arity $r$.
Let $t=\lambda y_1\ldots y_n.M[u]$ be closed, where $M$ is a context with
one hole of ground type which is not in the scope of any $\lambda$ of $M$
(so $\FV(u)\subseteq\{\vec y\}$), and let $\mathcal M$ be a set of
\emph{applicative} observations.  Then
$\mathcal M=\mathcal M_1\cup\bigcup_{\vec v}\mathcal M_{2,\vec v}$, where
$\vec v$ ranges over the finitely many tuples in $(\{u\}\cup\Val^0)^r$,
and
\begin{itemize}
\item $t\ObsEq^1_{\mathcal M_1}\lambda\vec y.M[M[u]]$;
\item $t\ObsEq^1_{\mathcal M_{2,\vec v}}\lambda\vec y.\,f\,v_1\ldots v_r$;
  moreover, if some $v_j=u$, then $\NF(u[\vec s/\vec y])\in\Val^0$ for
  every $\vec s\in\mathcal M_{2,\vec v}$.
\end{itemize}
For $i=0$ no splitting is needed: $t\ObsEq^0_{\mathcal M}\lambda\vec y.M[M[u]]$.
\end{proposition}
\begin{proof}
Let $x$ be a fresh variable of type $\Ground$.  By the assumption on the
hole, $M[u]=M[x][u/x]$ and $M[M[u]]=M[x][M[u]/x]$ without capture.
Fix $\vec s\in\mathcal M$ and write $\sigma=[\vec s/\vec y]$.  Then
$t\vec s=_\beta M[x]\sigma\,[u\sigma/x]$ and
$(\lambda\vec y.M[M[u]])\vec s=_\beta M[x]\sigma\,[M[u]\sigma/x]$, where
$u\sigma$ and $M[u]\sigma$ are closed of type $\Ground$.

$P=\NF(M[x]\sigma)$ is a ground normal form whose only free variable is
$x$; hence $P$ is $x$, a constant $a$, or $f(p_1,\ldots,p_r)$ with each
$p_j$ again such a normal form, and $\NF(M[x]\sigma\,[w/x])=P[\NF(w)/x]$
for every closed ground $w$.  Let $w=\NF(u\sigma)$.
\begin{itemize}
\item $P=x$.  Then $\NF(t\vec s)=w$ and
  $\NF(M[M[u]]\sigma)=\NF(M[u]\sigma)=\NF(t\vec s)$.  Put $\vec s$ into
  $\mathcal M_1$.
\item $P=a$.  Both sides normalize to $a$.  Put $\vec s$ into
  $\mathcal M_1$.
\item $P=f(p_1,\ldots,p_r)$.  If $\NF(t\vec s)\notin\Val^1$, then either
  some $p_j$ is neither $x$ nor a constant, or some $p_j=x$ and $w$ is not
  a constant, and then $\NF(M[u]\sigma)=P[w/x]$ is not a constant either;
  so $\NF(M[M[u]]\sigma)\notin\Val^1$, both sides are unobservable, and
  $\vec s$ goes into $\mathcal M_1$.  Otherwise every $p_j[w/x]$ is a
  constant, so each $p_j$ is a constant or $x$, and if some $p_j=x$, then
  $w\in\Val^0$.  Let $v_j=p_j$ if $p_j$ is a constant and $v_j=u$ if
  $p_j=x$, and put $\vec s$ into $\mathcal M_{2,\vec v}$.  Then
  $t\vec s=_\beta(\lambda\vec y.f\,v_1\ldots v_r)\vec s$.
\end{itemize}
If $i=0$, in the third case both sides start with $f$, so neither lies in
$\Val^0$, and all $\vec s$ go into $\mathcal M_1$.  Conclude by
Lemma~\ref{lem:basic}(c).
\end{proof}

\begin{remark}\label{rem:multicall}
The two restrictions in Proposition~\ref{prop:stretch} are essential for
its proof.  A non-applicative observation may call $t$ several times, and
the calls may fall into different cases; e.g.\ for
$t=\lambda y_1y_2.\,y_1y_2$ and $C[-]=[-]\,F\,([-]\,I\,a)$ with
$F=\lambda z.f\,z\,a$, $I=\lambda z.z$, neither replacement is equivalent
to $t$ on $C$.  Since the calculus has no destructors for ground values, no
single term can switch between the cases.  The same happens when the hole
of $M$ lies under a $\lambda$: then $u$ is evaluated several times within
one observation.  Both problems disappear in the semantic version below.
\end{remark}

\subsection{Semantic stretching}

Let $t=\lambda\vec y.M[u]$ with $M$ a context with one hole of type
$\Ground$, and let $\vec z$ be the variables bound in $M$ above the hole,
so $\FV(u)\subseteq\vec y\cup\vec z$.  Let $\tau$ be the type of
$\hat u=\lambda\vec y\vec z.u$, let $h:\tau$ be fresh and $\vec z'$ fresh
dummy variables of the types of $\vec z$.  Define $\Phi_M:D_\tau\to D_\tau$
by
\[
  \Phi_M(\varphi)=\sem{\lambda\vec y\vec z'.\,M[h\,\vec y\,\vec z]}_{h:=\varphi},
\]
and write $M^1=M$, $M^{k+1}[\,]=M[M^k[\,]]$.

\begin{proposition}[Semantic stretching]\label{prop:semstretch}
For every $k\ge1$, $\sem{\lambda\vec y\vec z'.M^k[u]}=\Phi_M^k(\sem{\hat u})$.
Consequently, if $\Phi_M^a=\Phi_M^b$ for some $1\le a<b$, then
$\sem{\lambda\vec y.M^a[u]}=\sem{\lambda\vec y.M^b[u]}$ for every $u$, hence
$\lambda\vec y.M^a[u]\ObsEq^i_{\mathcal M}\lambda\vec y.M^b[u]$ for every
$\mathcal M$.  Such $a<b\le|D_\tau|^{|D_\tau|}+1$ always exist.
\end{proposition}
\begin{proof}
For $k=1$: $M[u]=_\beta M[\hat u\,\vec y\,\vec z]$ and compositionality.
For $k+1$: $M^k[u]$ has free variables among $\vec y$ only, so it may be
placed in the hole of $M$ without capture, and
$\sem{\lambda\vec y\vec z'.M[M^k[u]]}=\Phi_M(\Phi_M^k(\sem{\hat u}))$.
The rest follows from Theorem~\ref{thm:semchar} and the pigeonhole
principle in the finite monoid $D_\tau^{D_\tau}$.
\end{proof}

Proposition~\ref{prop:stretch} describes $\Phi_M$ at a single argument:
identity ($P=x$), constant ($P=a$), or $f$ applied to the argument and
constants ($P=f(\ldots)$).

\section{Cells}\label{sec:cells}

\subsection{Slots}

Fix the type $A=A_1\to\cdots\to A_m\to\Ground$ of the solution.  Let
$T(A)$ be the smallest set of types containing $A_1,\ldots,A_m$ and
closed under taking argument types; it contains the types of all
variables bound in $\eta$-long normal forms of type $A$, and it is finite.

Variables are organised in \emph{slots}, each with a fixed type:
\begin{itemize}
\item the \emph{initial variables} $X=\{x_1,\ldots,x_m\}$, $x_l:A_l$;
\item for each $\sigma\in T(A)$, $n_\sigma$ slots with variables
  $Y_\sigma=\{y^\sigma_1,\ldots,y^\sigma_{n_\sigma}\}$ and fresh copies
  $Y'_\sigma=\{y'^\sigma_1,\ldots\}$.
\end{itemize}
Write $Y=\bigcup_\sigma Y_\sigma$, $Y'=\bigcup_\sigma Y'_\sigma$.  The
variables of $Y_\Ground$ are the \emph{level-0} variables (written
$z_1,\ldots,z_k$ earlier, $k=n_\Ground$), the others the \emph{level-1}
variables (earlier $n=\sum_{\sigma\neq\Ground}n_\sigma$).  Unprimed
variables are bound outside a cell, primed ones by the arguments of a cell.
Initial variables are bound only at the root and never rebound, so they
need no primed copies.  Let
\[
  \tau=A_1\to\cdots\to A_m\to\vec\sigma_Y\to\Ground
\]
be the type of $\lambda\vec x\vec y.M$ with $\vec y$ an enumeration of
$Y$.  All holes are of type $\Ground$.

\subsection{Cells and cellular terms}

\begin{definition}[Cell]
A \emph{cell} is a context of one of the forms
\[
\mcell\hole1\ldots\hole K=y\,(\lambda\vec w_1.\hole1)\cdots(\lambda\vec w_K.\hole K)
\qquad\text{or}\qquad
\mcell\hole1\ldots\hole r=f(\hole1,\ldots,\hole r),
\]
where $y\in X\cup Y$ and the binders $\vec w_j$ are taken from $Y'$
according to their types.  In particular each $\lambda\vec w_j.\hole j$ is
closed apart from the hole, and $y$ is the unique head of the cell.
Padovani's cells are the case $m=0$.
\end{definition}

\begin{definition}[Slot condition]
Let $b$ be a term of type $\Ground$ in hole $j$ of a cell, so that
$\FV(b)\subseteq X\cup Y\cup\vec w_j$.  It \emph{fits the slots} if its
free variables can be assigned slots of their types so that distinct
variables get distinct slots.  Renaming each free variable by the
unprimed variable of its slot gives $M$ with
\[
   b =_\beta (\lambda\vec x\vec y.M)\,\vec x\,y''_1\ldots,
   \qquad y''_l\in\{y_l,y'_l\},
\]
where $y''_l$ is primed iff slot $l$ is occupied by a variable of
$\vec w_j$.  We call $c=\lambda\vec x\vec y.M:\tau$ the \emph{closure} of
$b$ and $\rho=\vec x\,y''_1\ldots$ its \emph{slot vector}.
\end{definition}

Each slot delivers either the outer or the freshly bound variable, never
both: this is where the slot numbers bound the term.

\begin{definition}[Cellular closures and terms]\label{def:cellular}
The \emph{cellular closures} form the least set of closed terms of type
$\tau$ containing
\begin{itemize}
\item $\lambda\vec x\vec y.\,a$ and $\lambda\vec x\vec y.\,z$ for constants
  $a$ and $z\in Y_\Ground$ (the \emph{allowed right-hand sides});
\item $\lambda\vec x\vec y.\,\mcell[c_1\rho_1]\ldots[c_K\rho_K]$ for a cell
  $\mcell$, cellular closures $c_j$ and slot vectors $\rho_j$.
\end{itemize}
A closed $t=\lambda\vec x.M:A$ is \emph{cellular} if $\lambda\vec x\vec y.M$
is a cellular closure in which no unprimed $y\in Y$ occurs free.  The
$c_j\rho_j$ are $\beta$-redexes recording which variable occupies which
slot, so cellular terms are not normal.
\end{definition}

\subsection{Cellular terms form a regular set}

\begin{lemma}[Cells as finite operations]\label{lem:automaton}
For fixed slot numbers there is a finite ranked alphabet $\Gamma$, with
symbols $(\mcell,\rho_1,\ldots,\rho_K)$ of rank $K$ and the allowed
right-hand sides of rank $0$, such that cellular closures are exactly the
trees over $\Gamma$ and
\[
  \sem{\lambda\vec x\vec y.\,\mcell[c_1\rho_1]\ldots[c_K\rho_K]}
  =\hat\mcell_{\vec\rho}(\sem{c_1},\ldots,\sem{c_K}),
  \qquad
  \hat\mcell_{\vec\rho}(\vec\varphi)
  =\sem{\lambda\vec x\vec y.\,\mcell[h_1\rho_1]\ldots[h_K\rho_K]}_{h_j:=\varphi_j}.
\]
Thus $c\mapsto\sem c$ is computed by a deterministic bottom-up tree
automaton with the finite state set $D_\tau$.
\end{lemma}
\begin{proof}
The head of a cell is one of finitely many slot variables or symbols $f$;
the number of holes is bounded by the arity of its type; binders and slot
vectors range over finite sets.  The equation is compositionality.
\end{proof}

\begin{corollary}\label{cor:cellular-decidable}
For fixed slot numbers it is decidable whether a matching problem has a
cellular solution, and if it has one, it has one of depth at most
$|D_\tau|$.
\end{corollary}
\begin{proof}
By Theorem~\ref{thm:semchar} the cellular solutions are recognized by the
automaton of Lemma~\ref{lem:automaton} with accepting states
$\{\varphi\mid\varphi(\sem{\vec s_q},\vec d)=b_q\text{ for all }q\}$
(the values $\vec d$ of the unused $\vec y$ are irrelevant).  Emptiness
of tree automata is decidable, and a non-empty language contains a tree
of depth at most the number of states.
\end{proof}

The corollary bounds the depth without explicit pumping: the automaton
performs it, and Proposition~\ref{prop:semstretch} is the special case of
a single repeated layer.  It also shows that, once the slots are bounded,
minimal solutions are built from finitely many ``representative'' closures
(one per reachable state).

\section{Semi-cellular terms}\label{sec:semicell}

Let $t$ be closed with
$\NF(t)=\lambda\vec x.\,y\,(\lambda\vec w_1.b_1)\cdots(\lambda\vec w_K.b_K)$,
so $\NF(t)=\lambda\vec x.\mcell[b_1]\ldots[b_K]$ for its \emph{top cell}
$\mcell$.  If every $b_j$ fits the slots, with closure $c'_j$ and slot
vector $\rho_j$, we write for closed $e_1,\ldots,e_K:\tau$
\[
  t\{e_1,\ldots,e_K\}=\lambda\vec x.\,\mcell[e_1\rho_1]\ldots[e_K\rho_K],
\]
so $t=_\beta t\{c'_1,\ldots,c'_K\}$, and use the contexts and
\emph{induced observations}
\[
  D_j^{e_1\ldots e_{j-1}}[-]=t\{e_1,\ldots,e_{j-1},[-],c'_{j+1},\ldots,c'_K\},
  \qquad
  \mathcal M_j^{e_1\ldots e_{j-1}}=\mathcal M\circ D_j^{e_1\ldots e_{j-1}}.
\]
They describe how $\mathcal M$ observes hole $j$ once holes
$1,\ldots,j-1$ have been replaced by $e_1,\ldots,e_{j-1}$.  The same
notation applies to closures $c:\tau$ in place of $t$.

\begin{definition}[Semi-cellular under $\mathcal M$]\label{def:semicell}
By induction on the size of $\NF(t)$: $t$ is semi-cellular under
$\mathcal M$ if
\begin{enumerate}[label=(\roman*)]
\item $t\ObsEq^i_{\mathcal M}\lambda\vec x.a$ for an allowed right-hand
  side $a$; or
\item the top cell of $\NF(t)$ is a cell, all $b_j$ fit the slots, and for
  every $j\le K$ and all cellular $e_1,\ldots,e_{j-1}$ with
  $e_l\ObsEq^i_{\mathcal M_l^{e_1\ldots e_{l-1}}}c'_l$ $(l<j)$, the closure
  $c'_j$ is semi-cellular under $\mathcal M_j^{e_1\ldots e_{j-1}}$.
\end{enumerate}
\end{definition}

\begin{lemma}\label{lem:semicell}
If $t$ is semi-cellular under $\mathcal M$, there is a cellular $e$ with
$t\ObsEq^i_{\mathcal M}e$.
\end{lemma}
\begin{proof}
By induction on the size of $\NF(t)$.  In case~(i) take
$e=\lambda\vec x.a$.  In case~(ii) choose cellular $e_1,\ldots,e_K$ one
after another: given $e_1,\ldots,e_{j-1}$, the closure $c'_j$ is smaller
than $\NF(t)$ and semi-cellular under $\mathcal M_j^{e_1\ldots e_{j-1}}$,
so by induction there is a cellular $e_j$ with
$c'_j\ObsEq^i_{\mathcal M_j^{e_1\ldots e_{j-1}}}e_j$.  With
$t_j=t\{e_1,\ldots,e_j,c'_{j+1},\ldots,c'_K\}$ we have
$t_{j-1}=D_j^{e_1\ldots e_{j-1}}[c'_j]$ and
$t_j=D_j^{e_1\ldots e_{j-1}}[e_j]$, so Lemma~\ref{lem:congruence} gives
$t_{j-1}\ObsEq^i_{\mathcal M}t_j$, and
$t=_\beta t_0\ObsEq^i_{\mathcal M}\cdots\ObsEq^i_{\mathcal M}t_K$, which
is cellular.
\end{proof}

\begin{remark}
The holes are replaced one after another because an observation may pass
the closure of one hole inside an argument to another hole: replacing
$c'_l$ by an $e_l$ that agrees with it only on $\mathcal M_l$ changes what
$c'_j$ receives.  Even for applicative $\mathcal M$ the induced
$\mathcal M_j$ are in general not applicative (Remark~\ref{rem:multicall}).
\end{remark}

\begin{conjecture}[Main conjecture]\label{conj:main}
For every type $A$ and every finite applicative $\mathcal M$ at type $A$
there are computable slot numbers such that every closed $t:A$ is
semi-cellular under $\mathcal M$.
\end{conjecture}

By Lemma~\ref{lem:semicell}, Theorem~\ref{thm:semchar} and
Corollary~\ref{cor:cellular-decidable}, the conjecture implies
decidability of higher-order matching.  Equivalently: every
$\lambda$-definable $\sem t$ agrees on the points $\sem{\vec s_q}$ with
the denotation of a term using boundedly many live variables of each
type.  The conjecture cannot be proved in the full model alone: by
Loader's theorem $\lambda$-definability in finite full models is
undecidable, so the proof must use that only finitely many points matter.
Stirling's small-solution theorem implies the existential version
(some solution is cellular); an independent proof is what we are after.

\section{Towards the main conjecture: bounding the slots}\label{sec:bounding}

\subsection{Representatives and merging}\label{sec:merge}

Let $\Def_\sigma\subseteq D_\sigma$ be the set of $\lambda$-definable
elements.  For every $\varphi\in\Def_\sigma$ we fix a closed term
$R_\varphi$ with $\sem{R_\varphi}=\varphi$, taken as it is (a
\emph{representative}).  For a definable environment $\eta$ write $R_\eta$
for the substitution $v\mapsto R_{\eta(v)}$.  Every value that occurs
when a closed term is evaluated on closed arguments is definable.

\begin{lemma}[Representatives]\label{lem:repr}
If $\FV(u)\subseteq\vec v$ and $\eta$ is definable, then
$\sem u_\eta=\sem{u[R_\eta]}$; in particular $\sem{\lambda\vec w.u}_\eta$
is definable.
\end{lemma}

\begin{definition}[Definable equivalence]
$\approx_\Ground$ is equality on $\Def_\Ground$, and
$\varphi\approx_{\sigma\to\tau}\psi$ iff $\varphi\,a\approx_\tau\psi\,a'$
for all $a\approx_\sigma a'$ in $\Def_\sigma$.
\end{definition}

\begin{lemma}[Fundamental lemma]\label{lem:fundamental}
Every definable element is $\approx$-related to itself.  Hence, if $F$ is
definable and $\varphi_j\approx\psi_j$, then $F\,\vec\varphi=F\,\vec\psi$.
Moreover, definable $H,H':\vec\tau\to\Ground$ that agree on all definable
arguments satisfy $H\approx H'$.
\end{lemma}
\begin{proof}[Proof sketch]
The definable elements form a sub-applicative structure containing all
closed-term denotations, so the fundamental lemma of logical relations
applies.  For the last claim, $H\vec a=H\vec a'=H'\vec a'$ for
$\vec a\approx\vec a'$.
\end{proof}

Let $c=\lambda\vec v.\,y\,(\lambda\vec w_1.b_1)\cdots(\lambda\vec w_K.b_K)$
and let $E\subseteq\Def_{\vec v}$ be a finite set of environments at which
the value of $c$ matters.  The \emph{needed environments} of hole $j$ are
\[
  E_j=\{\,(\eta,\vec a)\mid\eta\in E,\ \vec a\in\Def_{\vec w_j}\,\},
  \qquad |E_j|\le|E|\cdot|D_{\vec w_j}|.
\]

\begin{lemma}[Local replacement]\label{lem:local}
If $\sem{b'_j}_e=\sem{b_j}_e$ for all $e\in E_j$ and all $j$, then
$c'=\lambda\vec v.\,y\,(\lambda\vec w_1.b'_1)\cdots(\lambda\vec w_K.b'_K)$
satisfies $\sem{c'}(\eta)=\sem c(\eta)$ for all $\eta\in E$.
\end{lemma}
\begin{proof}
For $\eta\in E$, $\sem c(\eta)=\eta(y)(H^\eta_1,\ldots,H^\eta_K)$ with
$H^\eta_j=\sem{\lambda\vec w_j.b_j}_\eta$ definable
(Lemma~\ref{lem:repr}).  $H'^\eta_j$ and $H^\eta_j$ agree on all definable
points, so they are $\approx$-related, and $\eta(y)$ is definable;
apply Lemma~\ref{lem:fundamental}.
\end{proof}

\begin{lemma}[Merging]\label{lem:merging}
If $v,v'$ are variables of $\vec v\cup\vec w_j$ of the same type with
$e(v)=e(v')$ for all $e\in E_j$, then $b'_j=b_j[v/v']$ satisfies the
hypothesis of Lemma~\ref{lem:local}.  Hence at hole $j$ one may assume at
most $|D_\sigma|^{|E_j|}$ live variables of type $\sigma$.
\end{lemma}
\begin{proof}
$\sem{b_j[v/v']}_e=\sem{b_j}_{e[v':=e(v)]}=\sem{b_j}_e$.
\end{proof}

Applied top-down from $E^{(0)}=\{\sem{\vec s_q}\}_q$, the lemmas preserve
the values on $E^{(0)}$, so the result is still a solution.  Passing a
closure inside an argument to another hole does no harm, because $E_j$
contains all definable points.  The bounds stay computable, since only
$|\Def_\sigma|\le|D_\sigma|$ is used, although $\Def_\sigma$ itself is not
computable.

\begin{example}[The environments multiply]\label{ex:multiply}
Let $x:(\Ground\to\Ground)\to\Ground$, $\mathcal M=\{s\}$ with
$s=\lambda g.\,f\,(g\,a)\,(g\,b)$, and
$t=\lambda x.\,x(\lambda z_1.\,x(\lambda z_2.\cdots x(\lambda z_d.\,M)\cdots))$.
Each level calls the next one twice, so $M$ is evaluated in $2^d$
environments, one for each $(z_1,\ldots,z_d)\in\{a,b\}^d$, and the value
vectors of $z_1,\ldots,z_d$ are pairwise distinct.  The bound of
Lemma~\ref{lem:merging} is $|D_\Ground|^{2^d}$: it depends on the depth.
In general $|E^{(d+1)}|\le|E^{(d)}|\cdot\max|D_{\vec w}|$, and counting
$|E^{(d)}|\le|D_{\mathrm{scope}}|$ instead is circular: $n\le|D|^{|D|^n}$
does not close.
\end{example}

\subsection{Contexts as transducers: intersection types}\label{sec:itypes}

In Section~\ref{sec:merge} a hole had to be correct on all definable
values of its bound variables, because in the full model a context is a
function of the value of its hole and does not record which points it
queries.  We now treat a context as a transducer that, from a requirement
on the whole term, produces the requirements it actually sends into its
holes.

\begin{definition}[Refinement types]
$\Ty_\Ground=D_\Ground$ and
$\Ty_{\sigma\to\tau}=\{\,S\to\theta\mid S\subseteq\Ty_\sigma,\ \theta\in\Ty_\tau\,\}$,
with $S$ read as an intersection ($S=\varnothing$ allowed) and
\[
  \sem d=\{d\},\qquad
  \sem{S\to\theta}=\{\,\varphi\mid
     \forall a\in\textstyle\bigcap_{s\in S}\sem s.\ \varphi\,a\in\sem\theta\,\}.
\]
A type environment $\Delta$ assigns to variables $x:\sigma$ sets
$\Delta(x)\subseteq\Ty_\sigma$.  The rules are
\[
\frac{\theta\in\Delta(x)}{\Delta\vdash x:\theta}
\qquad
\frac{}{\Delta\vdash a:a}
\qquad
\frac{}{\Delta\vdash f:\{d_1\}\to\cdots\to\{d_r\}\to\hat f(\vec d)}
\]
\[
\frac{\Delta,x:S\vdash M:\theta}{\Delta\vdash\lambda x.M:S\to\theta}
\qquad
\frac{\Delta\vdash M:S\to\theta\qquad\Delta\vdash N:s\ \text{for all } s\in S}
     {\Delta\vdash M\,N:\theta}
\]
\end{definition}

\begin{lemma}[Soundness and ground completeness]\label{lem:it}
If $\Delta\vdash M:\theta$ and $\eta(x)\in\bigcap_{s\in\Delta(x)}\sem s$
for all $x$, then $\sem M_\eta\in\sem\theta$.  For closed $N:\Ground$,
$\vdash N:d$ iff $\sem N=d$.  Hence $\sem{t\,\vec s_q}=b_q$ iff
$\vdash t:S_1\to\cdots\to S_n\to b_q$ for some $S_i$ with
$\vdash s_{q,i}:s$ for all $s\in S_i$.
\end{lemma}
\begin{proof}[Proof sketch]
Soundness by induction on derivations.  Completeness by subject expansion
(the argument of a contracted redex is typed at the types at which the
substituted variable was used, possibly none) and a direct derivation for
the first-order normal form.  The last claim is inversion of the
application rule.
\end{proof}

So a matching problem becomes a finite set of \emph{top requirements}
$\vdash t:\theta_q$.  For a context $L$ with one hole and variables
$\vec v$ in scope at the hole, a \emph{hole requirement} is a judgement
$\Delta\vdash\hole{}:d$ with $\mathrm{dom}(\Delta)\subseteq\vec v$;
$\vdash_J$ extends the rules by the axioms
$\Delta'\vdash\hole{}:d$ for $(\Delta\vdash\hole{}:d)\in J$ and
$\Delta\subseteq\Delta'$.

\begin{lemma}[Decomposition]\label{lem:decomp}
$\Gamma\vdash L[b]:\theta$ iff there is a finite set $J$ of hole
requirements with $\Gamma\vdash_J L:\theta$ and $\Delta\vdash b:d$ for all
$(\Delta\vdash\hole{}:d)\in J$.
\end{lemma}
\begin{proof}
Cut the derivation at the subderivations for $b$, respectively plug them
in, using admissible weakening.
\end{proof}

Thus $L$ is a nondeterministic transducer
$(\Gamma\vdash\theta)\mapsto\{J\mid\Gamma\vdash_J L:\theta\}$.  For a cell
$y\,(\lambda\vec w_1.\hole1)\cdots(\lambda\vec w_K.\hole K)$ it is explicit:
a requirement $\Gamma\vdash\mcell:d$ is met by choosing
$(S_1\to\cdots\to S_K\to d)\in\Gamma(y)$ and sending, for each
$(R_1\to\cdots\to R_l\to d')\in S_j$, the requirement
$\Gamma,\vec w_j:\vec R\vdash\hole j:d'$ into hole $j$.  Unlike $E_j$,
the set $J$ contains only what is actually used.

\begin{lemma}[Typed merging]\label{lem:typed-merge}
If $\Delta\vdash b:d$ for all $(\Delta\vdash\hole{}:d)\in J$ and
$\Delta(v')\subseteq\Delta(v)$ for all of them, then
$\Delta\vdash b[v/v']:d$ for all of them.  Hence at a hole with
requirement set $J$ one may assume at most
$(2^{|\Ty_\sigma|})^{|J|}$ live variables of type $\sigma$, and variables
with $\Delta(v)=\varnothing$ throughout are dead.
\end{lemma}
\begin{proof}
Replace each use of $v'$ by $v$; the type used for $v'$ lies in
$\Delta(v)$.  The term still meets its top requirements by
Lemma~\ref{lem:decomp}.
\end{proof}

\begin{conjecture}[Bounded requirements]\label{conj:req}
For every type $A$ and every finite set of top requirements there is a
computable $B$ such that every term meeting them does so via derivations
in which every hole of its cell decomposition receives at most $B$
distinct requirements.
\end{conjecture}

Conjecture~\ref{conj:req} and Lemma~\ref{lem:typed-merge} bound the live
variables, which gives Conjecture~\ref{conj:main}.

\section{Open problems and directions}\label{sec:open}

\begin{enumerate}
\item \textbf{The circularity.}  Requirements multiply along a path of
  cells (one per type in each $S_j$), but $J$ is a set, and after merging
  the number of distinct requirements is bounded by a function of the slot
  numbers.  The task is to break the circularity between the slot bound
  and the requirement bound (Example~\ref{ex:multiply} shows that counting
  coordinates alone does not).
\item \textbf{Counting distinctions instead of coordinates.}  In
  Example~\ref{ex:multiply} the variables $z_1,\ldots,z_d$ have distinct
  vectors, but a ground leaf of depth at most $1$ uses at most $r$ of them.
  The question is how many variables a subterm of higher type can carry
  further down, i.e.\ how many it actually distinguishes on its calls.
\item \textbf{Pumping on requirements.}  If two holes on a path receive the
  same requirement set (up to renaming of slots), the segment between them
  can be cut, as in Proposition~\ref{prop:semstretch} but for derivations.
  This removes repetitions but does not by itself bound $|J|$.
\item \textbf{Inherited requirements.}  Requirements on variables bound
  far above a hole are passed down unchanged; one may try to merge
  inherited variables at the level where they are bound.
\item \textbf{Minimal derivations.}  Conjecture~\ref{conj:req} only asks
  for good derivations to exist, so one may choose derivations with the
  fewest distinct requirements.
\item \textbf{Representatives of arguments.}  Replacing each argument of a
  cell by a fixed representative of its class bounds the size of a
  solution once the free variables of the arguments are bounded
  (Corollary~\ref{cor:cellular-decidable}); without such a bound the
  arguments are open terms whose closures have the order of $A$, so the
  induction on order does not apply.
\item \textbf{Technical TODOs.}  The context lemma in the preorder form
  (Lemma~\ref{lem:context}); full proofs of Lemmas~\ref{lem:fundamental}
  and~\ref{lem:it}.
\end{enumerate}
